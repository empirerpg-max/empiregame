function driveUrl(url) {
  if (!url) return "";
  const sUrl = String(url).trim();
  if (sUrl.indexOf('drive.google.com') > -1) {
    var id = "";
    if (sUrl.indexOf('id=') > -1) { id = sUrl.split('id=')[1].split('&')[0]; }
    else { var parts = sUrl.split('/'); for (var i=0;i<parts.length;i++){if(parts[i]==='d'){id=parts[i+1];break;}} }
    if (id) return "https://lh3.googleusercontent.com/d/" + id;
  }
  return sUrl;
}

    const wsLog = aba('QUERIDOMETRO_LOG');
    const logs = wsLog.getDataRange().getValues().slice(1);
    const votosSemana = logs.filter(l => l[6] === sem);
    const rankingMap = {};
    todosArtistas.forEach(a => { rankingMap[a.nome] = { nome: a.nome, img: a.foto, emojisRecebidos: [], totalVotos: 0 }; });
    votosSemana.forEach(v => { const para=v[2]; const emoji=v[3]; if(rankingMap[para]){rankingMap[para].emojisRecebidos.push(emoji);rankingMap[para].totalVotos+=1;} });
    const ranking = Object.values(rankingMap).sort((a,b)=>b.totalVotos-a.totalVotos);
    const configEmojis = [
      {emoji:'❤️',custo:30,label:'Amor'},{emoji:'🔥',custo:30,label:'Fogo'},{emoji:'⭐',custo:30,label:'Estrela'},
      {emoji:'👑',custo:30,label:'Coroa'},{emoji:'💀',custo:30,label:'Caveira'},{emoji:'🤡',custo:30,label:'Palhaço'},
      {emoji:'🐍',custo:30,label:'Cobra'},{emoji:'🥱',custo:30,label:'Tédio'}
    ];
    const reacoesPublicas = queridometro_mural_publico_(100);
    return { ok: true, meusArtistas: meusArtistasLocal, artistasAlvos: todosArtistas, configEmojis, ranking, semana: sem, reacoesPublicas };
  } catch(e) { return { ok: false, erro: e.message }; }

function queridometro_mural_publico_(limite) {
  limite = limite || 100;
  var ss = SpreadsheetApp.openById(ID_PRINCIPAL);
  var log = ss.getSheetByName("QUERIDOMETRO_LOG");
  if (!log || log.getLastRow() < 2) return [];
  var artistasSh = ss.getSheetByName("DB_ARTISTAS");
  var fotoPorNome = {};
  if (artistasSh && artistasSh.getLastRow() > 1) {
    var av = artistasSh.getRange(2,1,artistasSh.getLastRow()-1,3).getValues();
    av.forEach(function(r){if(r[0])fotoPorNome[String(r[0]).trim()]=r[1]||r[2]||"";});
  }
  var semanaAtual = getSemanaAno();
  var dados = log.getRange(2,1,log.getLastRow()-1,7).getValues();
  var out = [];
  for (var i=dados.length-1;i>=0&&out.length<limite;i--) {
    var row=dados[i]; var data=row[0]; var semanaLog=row[6];
    if(semanaLog!==semanaAtual)continue;
    var para=row[2]; var emoji=row[3];
    if(!para||!emoji)continue;
    var dataFormatada=data;
    if(data instanceof Date){dataFormatada=Utilities.formatDate(data,Session.getScriptTimeZone(),"dd/MM HH:mm");}
    else if(typeof data==='string'&&data.length>0){try{dataFormatada=Utilities.formatDate(new Date(data),Session.getScriptTimeZone(),"dd/MM HH:mm");}catch(e){}}
    out.push({para:String(para).trim(),fotoPara:fotoPorNome[String(para).trim()]||"",emoji:String(emoji),data:dataFormatada});
  }
  return out;
}

function handleQueridometroVotar(tgId, deNome, paraNome, emoji) {
  try {
    const sem = getSemanaAno();
    const configEmojis = [
      {emoji:'❤️',custo:30,pMin:5,pMax:10},{emoji:'🔥',custo:30,pMin:10,pMax:15},
      {emoji:'⭐',custo:30,pMin:15,pMax:20},{emoji:'👑',custo:30,pMin:18,pMax:20},
      {emoji:'💀',custo:30,pMin:-20,pMax:-15},{emoji:'🤡',custo:30,pMin:-15,pMax:-10},
      {emoji:'🐍',custo:30,pMin:-10,pMax:-5},{emoji:'🥱',custo:30,pMin:-2,pMax:2}
    ];
    const config = configEmojis.find(e => e.emoji === emoji);
    if (!config) return { ok: false, erro: 'Emoji inválido' };
    const wsArt = aba('DB_ARTISTAS');
    const data = wsArt.getDataRange().getValues();
    let rowDe=-1, rowPara=-1;
    for (let i=1;i<data.length;i++){if(data[i][0]===deNome)rowDe=i+1;if(data[i][0]===paraNome)rowPara=i+1;}
    if(rowDe===-1||rowPara===-1)return{ok:false,erro:'Erro ao localizar artistas'};
    const saldoDe=parseFloat(data[rowDe-1][3])||0;
    if(saldoDe<config.custo)return{ok:false,erro:'O artista '+deNome+' não tem saldo suficiente'};
    const prestigeGain=Math.floor(Math.random()*(config.pMax-config.pMin+1))+config.pMin;
    const prestigioPara=parseFloat(data[rowPara-1][8])||0;
    wsArt.getRange(rowDe,4).setValue(saldoDe-config.custo);
    wsArt.getRange(rowPara,9).setValue(prestigioPara+prestigeGain);
    const wsLog=aba('QUERIDOMETRO_LOG');
    if(wsLog.getLastRow()===0)wsLog.appendRow(['DATA','DE','PARA','EMOJI','CUSTO_PAGO','PRESTIGIO_GERADO','SEMANA']);
    wsLog.appendRow([new Date().toISOString(),deNome,paraNome,emoji,config.custo,prestigeGain,sem]);
    syncGameCoins(tgId,config.custo,0,'Queridômetro (Enviou '+emoji+' para '+paraNome+')',deNome);
    return { ok: true, msg: 'Ação enviada!' };
  } catch(e) { return { ok: false, erro: e.message }; }
}

// ── TOURS ──
function _lerVenuesDoSistema() {
  try {
    const wsVenues = aba('SISTEMA_TOURS');
    const data = wsVenues.getDataRange().getValues();
    if (data.length <= 1) return [];
    const venues = [];
    for (let i=1;i<data.length;i++){
      const row=data[i];
      if(!row[0]||!row[2])continue;
      venues.push({continente:String(row[0]||'').trim(),cidade:String(row[1]||'').trim(),local:String(row[2]||'').trim(),categoria:String(row[3]||'').trim(),capacidade:Number(row[4])||0,precoIngresso:Number(row[5])||0,repasseIngresso:Number(row[6])||0,lucroMaximo:Number(row[7])||0});
    }
    return venues;
  } catch(e) { Logger.log('Erro ao ler SISTEMA_TOURS: '+e.message); return []; }
}

function _obterVenuesPorContinente(continente) {
  return _lerVenuesDoSistema().filter(v=>v.continente.toLowerCase()===String(continente).toLowerCase());
}

function gerarAgenda(dataInicio, qtdShows, continente, tipoTourne) {
  const agenda = [];
  const startDate = new Date(dataInicio);
  const gapCfg = {'Indie':{gapMin:7,gapMax:12},'Arena':{gapMin:10,gapMax:18},'Estádio':{gapMin:14,gapMax:25}};
  const gc = gapCfg[tipoTourne] || gapCfg['Arena'];
  let venues = _obterVenuesPorContinente(continente);
  if (venues.length === 0) venues = _lerVenuesDoSistema();
  if (venues.length === 0) { Logger.log('CRÍTICO: Nenhum venue encontrado em SISTEMA_TOURS'); return []; }
  let dAtual = new Date(startDate);
  for (let i=0;i<Math.min(qtdShows,venues.length*2);i++){
    if(i>0){const gap=gc.gapMin+Math.floor(Math.random()*(gc.gapMax-gc.gapMin));dAtual=new Date(dAtual.getTime()+gap*86400000);}
    const dia=String(dAtual.getDate()).padStart(2,'0');
    const mes=String(dAtual.getMonth()+1).padStart(2,'0');
    const ano=dAtual.getFullYear();
    const venueIdx=i%venues.length;
    const venue=venues[venueIdx];
    const percentualOcupacao=60+Math.floor(Math.random()*38);
    const vendidos=Math.floor((venue.capacidade*percentualOcupacao)/100);
    agenda.push({numero:i+1,data:`${dia}/${mes}/${ano}`,local:venue.local,cidade:venue.cidade,categoria:venue.categoria,capacidade:venue.capacidade,vendidos,precoIngresso:venue.precoIngresso,repasseIngresso:venue.repasseIngresso,lucroMaximo:venue.lucroMaximo,receita:vendidos*venue.repasseIngresso,status:'Agendado'});
  }
  return agenda;
}

function handleCompraTour(p) {
  const precos = {'Indie':1000000,'Arena':5000000,'Estádio':15000000};
  const custo = precos[p.tipo] || 1000000;
  const ws = aba('DB_ARTISTAS');
  const data = ws.getDataRange().getValues();
  let artIdx = -1;
  for(let i=1;i<data.length;i++){if(String(data[i][0]).toLowerCase()===p.nome.toLowerCase()){artIdx=i+1;break;}}
  if(artIdx===-1)return{ok:false,erro:'Artista não encontrado'};
  const saldo=parseFloat(data[artIdx-1][3])||0;
  if(saldo<custo)return{ok:false,erro:'Saldo insuficiente'};
  ws.getRange(artIdx,4).setValue(saldo-custo);
  ws.getRange(artIdx,3).setValue('Turnê: '+p.titulo);
  const agenda=gerarAgenda(p.dataInicio,p.qtd,p.continente,p.tipo);
  if(agenda.length===0)return{ok:false,erro:'Nenhum venue encontrado. Adicione venues em SISTEMA_TOURS para o continente: '+p.continente};
  const totalEsperado=agenda.reduce((sum,s)=>sum+s.lucroMaximo,0);
  const tourInfo={titulo:p.titulo,tipo:p.tipo,qtd:p.qtd,continente:p.continente,shows_realizados:0,status:"Em andamento",totalEsperado,agenda};
  ws.getRange(artIdx,14).setValue(JSON.stringify(tourInfo));
  aba('CONTROLE_TOURS').appendRow([p.nome,p.titulo,p.tipo,p.qtd,0,p.continente||'Mundial',0,'Em andamento',JSON.stringify(agenda),'']);
  aba('RADAR_FEED').insertRowBefore(2).getRange(2,1,1,4).setValues([[new Date().toISOString(),p.nome,'Iniciou a turnê '+p.titulo,data[artIdx-1][1]]]);
  return{ok:true,message:'Turnê criada com '+agenda.length+' shows! Receita esperada: '+totalEsperado};
}

    for(let i=1;i<data.length;i++){
      const rowId=String(data[i][10]).trim();
      if(rowId===cleanId){if(!artistName||data[i][0]===artistName){userRow=i+1;break;}}
    }
    if(userRow===-1)return{ok:false,erro:'Artista não encontrado'};
    const saldoAnterior=parseFloat(data[userRow-1][3])||0;
    const vWager=parseFloat(wager||0);
    const vWon=parseFloat(won||0);
    if(vWager>saldoAnterior)return{ok:false,erro:'Saldo insuficiente'};
    const novoSaldo=saldoAnterior-vWager+vWon;
    ws.getRange(userRow,4).setValue(novoSaldo);
    const wsLog=aba('LOG_ECONOMIA');
    if(wsLog.getLastRow()===0)wsLog.appendRow(['DATA','TELEGRAM_ID','ARTISTA','TIPO','VALOR','CONTEXTO','SALDO_RESULTANTE']);
    if(vWager>0)wsLog.appendRow([new Date().toISOString(),cleanId,data[userRow-1][0],'DÉBITO',vWager,context||'Jogo',novoSaldo+vWon]);
    if(vWon>0)wsLog.appendRow([new Date().toISOString(),cleanId,data[userRow-1][0],'CRÉDITO',vWon,context||'Jogo',novoSaldo]);
    return{ok:true,novoSaldo};
  } catch(e){return{ok:false,erro:e.message};}
}

function handleSavePetState(tgId, payload) {
  try {
    if(!tgId||tgId==='guest')return{ok:false};
    const ws=aba('PET_STATE');
    if(ws.getLastRow()===0)ws.appendRow(['telegram_id','payload','last_update']);
    const data=ws.getDataRange().getValues();
    let rowIdx=-1;
    for(let i=1;i<data.length;i++){if(String(data[i][0]).trim()===String(tgId).trim()){rowIdx=i+1;break;}}
    const row=[tgId,payload,Date.now()];
    if(rowIdx!==-1)ws.getRange(rowIdx,1,1,3).setValues([row]);
    else ws.appendRow(row);
    return{ok:true};
  } catch(e){return{ok:false,erro:e.message};}
}

function getSemanaAno() {
  const d=new Date(); d.setHours(0,0,0,0); d.setDate(d.getDate()+4-(d.getDay()||7));
  const yearStart=new Date(d.getFullYear(),0,1);
  const weekNo=Math.ceil((((d-yearStart)/86400000)+1)/7);
  return d.getFullYear()+'-W'+weekNo;
}
