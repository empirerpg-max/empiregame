// ---------- helpers ----------
function _tv_col(headers, name) {
  var n = String(name).trim().toLowerCase();
  for (var i = 0; i < headers.length; i++) {
    if (String(headers[i]).trim().toLowerCase() === n) return i;
  }
  return -1;
}

function _tv_nowSP() {
  return new Date(new Date().toLocaleString('en-US', { timeZone: 'America/Sao_Paulo' }));
}

function _tv_parseDH(dataVal, horarioVal) {
  try {
    var year, month, day;
    if (dataVal instanceof Date) {
      year = dataVal.getUTCFullYear();
      month = dataVal.getUTCMonth() + 1;
      day = dataVal.getUTCDate();
    } else {
      var s = String(dataVal).trim();
      var iso = s.match(/^(\d{4})-(\d{2})-(\d{2})/);
      if (iso) { year = +iso[1]; month = +iso[2]; day = +iso[3]; }
      else {
        var br = s.match(/^(\d{2})\/(\d{2})\/(\d{4})/);
        if (!br) return null;
        day = +br[1]; month = +br[2]; year = +br[3];
      }
    }
    var hh = 0, mm = 0, sc = 0;
    if (horarioVal instanceof Date) {
      hh = horarioVal.getUTCHours();
      mm = horarioVal.getUTCMinutes();
      sc = horarioVal.getUTCSeconds();
    } else {
      var t = String(horarioVal).trim();
      var isoT = t.match(/T(\d{2}):(\d{2}):(\d{2})/);
      if (isoT) { hh = +isoT[1]; mm = +isoT[2]; sc = +isoT[3]; }
      else {
        var tm = t.match(/^(\d{1,2}):(\d{2})(?::(\d{2}))?/);
        if (tm) { hh = +tm[1]; mm = +tm[2]; sc = tm[3] ? +tm[3] : 0; }
      }
    }
    var pad = function (n) { return String(n).padStart(2, '0'); };
    return new Date(year + '-' + pad(month) + '-' + pad(day) +
      'T' + pad(hh) + ':' + pad(mm) + ':' + pad(sc) + '-03:00');
  } catch (e) { return null; }
}

function _tv_fmt(dt, pattern) {
  return Utilities.formatDate(dt, 'America/Sao_Paulo', pattern);
}

// ---------- listar_programas_tv ----------
function tv_listarProgramas() {
  var sh = aba('Agenda_TV');
  var rows = sh.getDataRange().getValues();
  if (rows.length <= 1) return [];

  var headers = rows[0];
  var iProg = _tv_col(headers, 'Programa');
  var iTipo = _tv_col(headers, 'Tipo');
  var iMat  = _tv_col(headers, 'Material');
  var iBuff = _tv_col(headers, 'Buff');
  var iData = _tv_col(headers, 'Data');
  var iHor  = _tv_col(headers, 'Horario');
  var iCapa = _tv_col(headers, 'Capa_URL');
  var iStat = _tv_col(headers, 'Status');
  var iTId  = _tv_col(headers, 'Topico_ID');
  var iTUrl = _tv_col(headers, 'Topico_URL');

  var now = _tv_nowSP();
  var items = [];

  for (var r = 1; r < rows.length; r++) {
    var row = rows[r];
    if (!row[iProg] && !row[iData]) continue;
    var startDt = _tv_parseDH(row[iData], row[iHor]);
    if (!startDt) continue;
    var statusLow = String(row[iStat] || '').trim().toLowerCase();
    items.push({
      _row: r + 1,
      _start: startDt,
      programa: String(row[iProg] || 'Empire TV').trim(),
      tipo: String(row[iTipo] || '').trim(),
      material: String(row[iMat] || '').trim(),
      buff: String(row[iBuff] || '').trim(),
      capa: String(row[iCapa] || '').trim(),
      status: statusLow,
      topicoId: String(row[iTId] || '').trim(),
      topicoUrl: String(row[iTUrl] || '').trim()
    });
  }
  items.sort(function (a, b) { return a._start - b._start; });

  // Detecta o programa "ao vivo" pela janela de tempo + Status
  var MAX_LIVE = 3 * 60 * 60 * 1000;
  var liveRow = -1;
  for (var i = 0; i < items.length; i++) {
    var it = items[i];
    var diff = now - it._start;
    if (diff >= 0) {
      var nextStart = items[i + 1] ? items[i + 1]._start : null;
      var until = nextStart || new Date(it._start.getTime() + MAX_LIVE);
      if (now < until) { liveRow = i; break; }
    }
  }

  return items.map(function (it, idx) {
    var ehLive = idx === liveRow ||
      it.status === 'transmitindo' || it.status === 'ao vivo' || it.status === 'live';
    var fin = it.status === 'finalizado' || it.status === 'concluido' || it.status === 'concluído';
    var data = _tv_fmt(it._start, 'dd/MM/yyyy');
    var hora = _tv_fmt(it._start, 'HH:mm');
    var id = it.topicoId || (it.programa + '_' + data + '_' + hora).replace(/\s+/g, '_');
    return {
      id: id,
      titulo: it.programa,
      subtitulo: it.material || it.tipo || '',
      categoria: it.tipo || '',
      ao_vivo: ehLive,
      finalizado: fin,
      status: it.status ? it.status.charAt(0).toUpperCase() + it.status.slice(1) : '',
      espectadores: 0,
      cover: it.capa || '',
      stream_url: it.topicoUrl || '',
      data: data,
      horario: hora,
      data_inicio: data + ' ' + hora,
      buff: it.buff || '',
      topico_id: it.topicoId || '',
      topico_url: it.topicoUrl || ''
    };
  });
}

// ---------- Presenca_TV ----------
function _tv_presSheet() {
  var sh = aba('Presenca_TV');
  if (sh.getLastRow() === 0) {
    sh.getRange(1, 1, 1, 5).setValues([['programa_id', 'telegram_id', 'nome', 'watched_seconds', 'last_seen']]);
    sh.getRange(1, 1, 1, 5).setBackground('#7c3aed').setFontColor('#ffffff').setFontWeight('bold');
  }
  return sh;
}

function tv_registrarPresenca(p) {
  var programa_id = String(p.programa_id || '');
  var telegram_id = String(p.telegram_id || p.tgId || '');
  var nome = String(p.nome || 'Anônimo');
  var watched = Number(p.watched_seconds || 0);
  if (!programa_id || !telegram_id) return { ok: false, error: 'params' };

  var sh = _tv_presSheet();
  var data = sh.getDataRange().getValues();
  var now = new Date();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]) === programa_id && String(data[i][1]) === telegram_id) {
      var prev = Number(data[i][3] || 0);
      sh.getRange(i + 1, 3).setValue(nome);
      sh.getRange(i + 1, 4).setValue(Math.max(prev, watched));
      sh.getRange(i + 1, 5).setValue(now);
      return { ok: true, updated: true };
    }
  }
  sh.appendRow([programa_id, telegram_id, nome, watched, now]);
  return { ok: true, created: true };
}

function tv_listarPresenca(p) {
  var programa_id = String(p.programa_id || '');
  if (!programa_id) return [];
  var sh = _tv_presSheet();
  var data = sh.getDataRange().getValues();
  var maxW = 0;
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]) === programa_id) {
      var w = Number(data[i][3] || 0);
      if (w > maxW) maxW = w;
    }
  }
  var out = [];
  for (var j = 1; j < data.length; j++) {
    if (String(data[j][0]) !== programa_id) continue;
    var ws = Number(data[j][3] || 0);
    out.push({
      telegram_id: String(data[j][1] || ''),
      nome: String(data[j][2] || 'Anônimo'),
      watched_seconds: ws,
      percentual: maxW > 0 ? Math.min(100, (ws / maxW) * 100) : 0
    });
  }
  return out;
}

// ---------- Chat_Arquivos ----------
function _tv_chatSheet() {
  var sh = aba('Chat_Arquivos');
  if (sh.getLastRow() === 0) {
    sh.getRange(1, 1, 1, 5).setValues([['Data', 'Hora', 'Sala', 'Total_Msgs', 'JSON_Completo']]);
    sh.getRange(1, 1, 1, 5).setBackground('#7c3aed').setFontColor('#ffffff').setFontWeight('bold');
  }
  return sh;
}

function tv_salvarChat(p) {
  var sala = String(p.sala || p.programa_id || '');
  var total = Number(p.total_msgs || 0);
  var json = String(p.json || '[]');
  if (!sala) return { ok: false, error: 'sala' };

  var sh = _tv_chatSheet();
  var data = sh.getDataRange().getValues();
  var tz = Session.getScriptTimeZone();
  var now = new Date();
  var dataStr = Utilities.formatDate(now, tz, 'yyyy-MM-dd');
  var horaStr = Utilities.formatDate(now, tz, 'HH:mm:ss');

  for (var i = 1; i < data.length; i++) {
    if (String(data[i][2]) === sala) {
      sh.getRange(i + 1, 1, 1, 5).setValues([[dataStr, horaStr, sala, total, json]]);
      return { ok: true, updated: true };
    }
  }
  sh.appendRow([dataStr, horaStr, sala, total, json]);
  return { ok: true, created: true };
}

function tv_listarArquivo() {
  var sh = aba('Chat_Arquivos');
  if (sh.getLastRow() < 2) return [];
  var data = sh.getDataRange().getValues();
  var out = [];
  for (var i = 1; i < data.length; i++) {
    if (!data[i][2]) continue;
    out.push({
      data: String(data[i][0] || ''),
      hora: String(data[i][1] || ''),
      sala: String(data[i][2] || ''),
      total_msgs: Number(data[i][3] || 0)
    });
  }
  return out.reverse();
}
