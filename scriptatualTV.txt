function doGet(e) {
  try {
    var acao     = e && e.parameter && e.parameter.acao ? e.parameter.acao : "";
    var callback = e && e.parameter && e.parameter.callback ? e.parameter.callback : null;

    if (acao === "tv_chat_list") return respondJson(handleChatListData(), callback);
    if (acao === "kick_fila")   return respondJson(getKickFila(), callback);

    var ss = SPREADSHEET_ID
      ? SpreadsheetApp.openById(SPREADSHEET_ID)
      : SpreadsheetApp.getActiveSpreadsheet();

    var agendaSheet    = ss.getSheetByName("Agenda_TV");
    var agendaTimeline = agendaSheet ? buildTimelineAgenda(agendaSheet) : [];

    // Programacao_RPG NÃO entra mais no fullSchedule — apenas Agenda_TV.
    var fullSchedule = agendaTimeline
      .sort(function(a, b) { return (a.inicio || 0) - (b.inicio || 0); });

    if (fullSchedule.length === 0) {
      return respondJson({
        status: "ok",
        message: "Sem programação cadastrada.",
        current: null,
        fullSchedule: []
      }, callback);
    }

    var current = findCurrent(fullSchedule);
    return respondJson({
      status:       "success",
      timestamp:    new Date().toISOString(),
      current:      current,
      fullSchedule: fullSchedule
    }, callback);

  } catch (err) {
    return respondJson({ status: "error", message: err.toString() }, null);
  }
}

function doPost(e) {
  try {
    var body = JSON.parse(e.postData.contents);
    var acao = body.acao || "";
    if (acao === "tv_chat_send")    return handleChatSend(body);
    if (acao === "tv_participacao") return handleParticipacao(body);
    if (acao === "kick_status")     return createJsonResponse(setKickStatus(body));
    return createJsonResponse({ ok: false, erro: "Ação desconhecida: " + acao });
  } catch (err) {
    return createJsonResponse({ ok: false, erro: err.toString() });
  }
}

function respondJson(data, callback) {
  var json = JSON.stringify(data);
  if (callback) {
    return ContentService
      .createTextOutput(callback + "(" + json + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService
    .createTextOutput(json)
    .setMimeType(ContentService.MimeType.JSON);
}

function createJsonResponse(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * Roda a cada 1 minuto (via trigger).
 * Verifica se existe algum item na fila cuja data/horário chegou,
 * e aciona o worker do Render se ainda não estiver transmitindo.
 */
function verificarEAcionarWorker() {
  try {
    var ss   = SPREADSHEET_ID ? SpreadsheetApp.openById(SPREADSHEET_ID) : SpreadsheetApp.getActiveSpreadsheet();
    var sh   = ss.getSheetByName("Programacao_RPG");
    if (!sh) return;

    var rows     = sh.getDataRange().getValues();
    var agora    = new Date();
    var hojeStr  = Utilities.formatDate(agora, "America/Sao_Paulo", "dd/MM/yyyy");
    var nowSecs  = getSecondsToday();
    var MARGEM   = 5 * 60; // aciona até 5 minutos após o horário

    var deveAcionar = false;

    for (var i = 1; i < rows.length; i++) {
      var driveId   = parseDriveId(rows[i][COL.DRIVE_ID]);
      if (!driveId) continue;

      var status    = String(rows[i][COL.STATUS] || "").trim().toLowerCase();
      if (status === "finalizado" || status === "transmitindo") continue;

      var dataItem    = formatarData(rows[i][COL.DATA]);
      if (dataItem && dataItem !== hojeStr) continue; // só itens de hoje

      var horarioSecs = parseHorarioToSeconds(rows[i][COL.HORARIO]);
      if (horarioSecs === null) continue;

      // Se o horário já passou (mas não mais de MARGEM minutos)
      if (nowSecs >= horarioSecs && nowSecs <= horarioSecs + MARGEM) {
        deveAcionar = true;
        break;
      }
    }

    if (!deveAcionar) {
      Logger.log("[Worker] Nenhum programa no horário agora.");
      return;
    }

    // Verifica se já está transmitindo
    var statusResp = UrlFetchApp.fetch(WORKER_URL + "/health", {
      muteHttpExceptions: true,
      method: "get"
    });
    var statusJson = JSON.parse(statusResp.getContentText());
    if (statusJson.status === "transmitindo") {
      Logger.log("[Worker] Já está transmitindo, não aciona novamente.");
      return;
    }

    // Aciona!
    var resp = UrlFetchApp.fetch(WORKER_URL + "/transmitir", {
      muteHttpExceptions: true,
      method: "get"
    });
    Logger.log("[Worker] Acionado: " + resp.getContentText());

  } catch (err) {
    Logger.log("[Worker] Erro em verificarEAcionarWorker: " + err);
  }
}

/**
 * Instala o trigger de 1 minuto para verificarEAcionarWorker.
 * Execute esta função UMA VEZ manualmente no Apps Script.
 */
function instalarTriggerWorker() {
  // Remove triggers antigos da mesma função
  ScriptApp.getProjectTriggers().forEach(function(t) {
    if (t.getHandlerFunction() === "verificarEAcionarWorker") ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger("verificarEAcionarWorker")
    .timeBased()
    .everyMinutes(1)
    .create();
  Logger.log("✅ Trigger do worker instalado! Verificando a cada 1 minuto.");
}

/**
 * Remove o trigger (use para pausar o acionamento automatico).
 */
function removerTriggerWorker() {
  ScriptApp.getProjectTriggers().forEach(function(t) {
    if (t.getHandlerFunction() === "verificarEAcionarWorker") ScriptApp.deleteTrigger(t);
  });
  Logger.log("⏹️ Trigger do worker removido.");
}

function onEdit(e) {
  try {
    var sheet = e.range.getSheet();
    if (sheet.getName() !== "Agenda_TV") return;
    var row = e.range.getRow();
    if (row <= 1) return;
    var rowData  = sheet.getRange(row, 1, 1, 10).getValues()[0];
    var programa = String(rowData[AGENDA_COL.PROGRAMA] || "").trim();
    var tipo     = String(rowData[AGENDA_COL.TIPO]     || "").trim();
    var material = String(rowData[AGENDA_COL.MATERIAL] || "").trim();
    var buff     = String(rowData[AGENDA_COL.BUFF]     || "").trim();
    var data     = formatarData(rowData[AGENDA_COL.DATA]);
    var horario  = formatarHorario(rowData[AGENDA_COL.HORARIO]);
    var topicoId = String(rowData[AGENDA_COL.TOPICO_ID] || "").trim();
    if (!programa || topicoId) return;
    var topico = postarNoCanalEObterTopico(programa, tipo, material, buff, data, horario);
    if (topico) {
      sheet.getRange(row, AGENDA_COL.STATUS     + 1).setValue("Agendado");
      sheet.getRange(row, AGENDA_COL.TOPICO_ID  + 1).setValue(topico.threadId);
      sheet.getRange(row, AGENDA_COL.TOPICO_URL + 1).setValue(topico.topicoUrl);
      Logger.log("✅ Tópico criado: " + topico.topicoUrl);
    }
  } catch (err) {
    Logger.log("Erro onEdit: " + err);
  }
}

function formatarData(val) {
  if (!val) return "";
  if (val instanceof Date) return Utilities.formatDate(val, "America/Sao_Paulo", "dd/MM/yyyy");
  return String(val).trim();
}

function formatarHorario(val) {
  if (!val) return "";
  if (val instanceof Date) return Utilities.formatDate(val, "America/Sao_Paulo", "HH:mm");
  return String(val).trim();
}

function getBotToken() {
  return PropertiesService.getScriptProperties().getProperty("BOT_TOKEN") || "";
}

function postarNoCanalEObterTopico(programa, tipo, material, buff, data, horario) {
  var token = getBotToken();
  if (!token) { Logger.log("BOT_TOKEN não configurado"); return null; }
  var linhas = [
    "📺 *" + programa + "*",
    data     ? "📅 " + data + (horario ? " às " + horario : "") : null,
    tipo     ? "🎬 " + tipo     : null,
    material ? "🎵 " + material : null,
    buff     ? "⚡ Buff: " + buff : null,
    "",
    "💬 Comente abaixo!"
  ].filter(function(l) { return l !== null; }).join("\n");
  var resp = UrlFetchApp.fetch(
    "https://api.telegram.org/bot" + token + "/sendMessage",
    {
      method: "post",
      contentType: "application/json",
      muteHttpExceptions: true,
      payload: JSON.stringify({ chat_id: CANAL_ID, text: linhas, parse_mode: "Markdown" })
    }
  );
  var result = JSON.parse(resp.getContentText());
  if (!result.ok) { Logger.log("Erro ao postar: " + JSON.stringify(result)); return null; }
  var messageId = result.result.message_id;
  Utilities.sleep(3000);
  var threadId = buscarThreadId(messageId);
  if (!threadId) { Logger.log("thread_id não encontrado para message_id " + messageId); return null; }
  return {
    threadId:  threadId,
    topicoUrl: "https://t.me/empireventos1/" + threadId,
    messageId: messageId
  };
}

function buscarThreadId(canalMessageId) {
  var token = getBotToken();
  for (var tentativa = 0; tentativa < 5; tentativa++) {
    Utilities.sleep(2000);
    var resp = UrlFetchApp.fetch(
      "https://api.telegram.org/bot" + token + "/getUpdates?limit=50&offset=-1",
      { muteHttpExceptions: true }
    );
    var data = JSON.parse(resp.getContentText());
    if (!data.ok) continue;
    for (var i = data.result.length - 1; i >= 0; i--) {
      var msg = data.result[i].message;
      if (!msg) continue;
      if (
        msg.chat && String(msg.chat.id) === String(SUPERGRUPO_ID) &&
        msg.is_automatic_forward &&
        msg.forward_from_message_id === canalMessageId
      ) {
        Logger.log("✅ thread_id: " + msg.message_id);
        return msg.message_id;
      }
    }
  }
  return null;
}

function handleParticipacao(body) {
  if (!body.tgId || !body.nome || !body.programa)
    return createJsonResponse({ ok: false, erro: "Dados incompletos" });
  var sheet = getParticipacaoSheet();
  var hoje  = Utilities.formatDate(new Date(), "America/Sao_Paulo", "dd/MM/yyyy");
  var rows  = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][3]) === String(body.tgId) &&
        String(rows[i][0]) === String(body.programa) &&
        String(rows[i][4]) === hoje) {
      sheet.getRange(i+1, 6).setValue((parseInt(rows[i][5]) || 0) + 1);
      recalcularPorcentagens(sheet, body.programa, hoje);
      return createJsonResponse({ ok: true });
    }
  }
  sheet.appendRow([body.programa, body.tipo||"", body.nome, body.tgId, hoje, 1, ""]);
  recalcularPorcentagens(sheet, body.programa, hoje);
  return createJsonResponse({ ok: true });
}

function recalcularPorcentagens(sheet, programa, data) {
  var rows = sheet.getDataRange().getValues();
  var total = 0, idxs = [];
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === programa && String(rows[i][4]) === data) {
      total += parseInt(rows[i][5]) || 0;
      idxs.push(i+1);
    }
  }
  if (total === 0) return;
  idxs.forEach(function(rowNum) {
    var msgs = parseInt(sheet.getRange(rowNum, 6).getValue()) || 0;
    sheet.getRange(rowNum, 7).setValue(Math.round((msgs/total)*100) + "%");
  });
}

function parseHorarioToSeconds(val) {
  if (!val || val === "") return null;
  if (val instanceof Date) {
    var s = Utilities.formatDate(val, "America/Sao_Paulo", "HH:mm:ss");
    var p = s.split(":");
    return (+p[0]*3600)+(+p[1]*60)+(+p[2]);
  }
  var str = String(val).trim();
  if (!str) return null;
  var m1 = str.match(/^(\d{1,2}):(\d{2})(?::(\d{2}))?$/);
  if (m1) return (+m1[1]*3600)+(+m1[2]*60)+(+(m1[3]||0));
  var m2 = str.match(/\d{4}-\d{2}-\d{2}[\sT](\d{1,2}):(\d{2})(?::(\d{2}))?/);
  if (m2) return (+m2[1]*3600)+(+m2[2]*60)+(+(m2[3]||0));
  var m3 = str.match(/\d{2}\/\d{2}\/\d{4}\s+(\d{1,2}):(\d{2})(?::(\d{2}))?/);
  if (m3) return (+m3[1]*3600)+(+m3[2]*60)+(+(m3[3]||0));
  return null;
}

function parseDriveId(val) {
  if (!val) return "";
  var str = String(val).trim();
  if (!str) return "";
  if (str.includes("drive.google.com") || str.startsWith("http")) {
    var m = str.match(/\/d\/([a-zA-Z0-9_-]{10,}?)(?:\/|\?|$)/i);
    return m ? m[1] : "";
  }
  if (/^[a-zA-Z0-9_-]{10,}$/.test(str)) return str;
  return "";
}

function buildTimeline(rows) {
  var dataRows = rows.slice(1);
  var timeline = [];
  var cursor   = 0;
  var pad   = function(n) { return String(Math.floor(n)).padStart(2,"0"); };
  var toHMS = function(s) { return pad(s/3600)+":"+pad((s%3600)/60)+":"+pad(s%60); };

  dataRows.forEach(function(row, idx) {
    var driveId   = parseDriveId(row[COL.DRIVE_ID]);
    if (!driveId) return;
    var statusRaw = String(row[COL.STATUS]||"" ).trim().toLowerCase();
    if (statusRaw === "finalizado") return;
    var duracao   = parseInt(row[COL.DURACAO], 10);
    if (isNaN(duracao) || duracao <= 0) return;
    var horarioSecs = parseHorarioToSeconds(row[COL.HORARIO]);
    var inicio = horarioSecs !== null ? Math.max(horarioSecs, cursor) : cursor;
    var fim    = inicio + duracao;
    cursor     = fim;
    var videoUrl = driveId.startsWith("http")
      ? driveId
      : PROXY_URL + "/video?file=video_" + driveId + ".mp4&secret=" + WORKER_SECRET;
    timeline.push({
      rowNum:     idx+2,
      driveId:    driveId,
      videoUrl:   videoUrl,
      inicio:     inicio,
      fim:        fim,
      duracao:    duracao,
      horarioStr: toHMS(inicio),
      data:       formatarData(row[COL.DATA]),
      programa:   String(row[COL.PROGRAMA]  ||"Empire TV").trim(),
      tipo:       String(row[COL.TIPO]      ||"").trim(),
      material:   String(row[COL.MATERIAL]  ||"").trim(),
      buff:       String(row[COL.BUFF]      ||"").trim(),
      topicoId:   String(row[COL.TOPICO_ID]  ||"").trim(),
      topicoUrl:  String(row[COL.TOPICO_URL] ||"").trim(),
      capaUrl:    String(row[COL.CAPA_URL]   ||"").trim(),
    });
  });
  return timeline;
}

function findCurrent(timeline) {
  if (timeline.length === 0)
    return { status:"off", programa:"Empire TV", message:"Nenhum programa agendado.", isBackup:true };
  var now = getSecondsToday();
  for (var i = 0; i < timeline.length; i++) {
    if (now >= timeline[i].inicio && now < timeline[i].fim)
      return buildPayload(timeline[i], now - timeline[i].inicio);
  }
  for (var i = 0; i < timeline.length; i++) {
    if (timeline[i].inicio > now) {
      return {
        status:"upcoming", rowNum:timeline[i].rowNum,
        programa:timeline[i].programa, tipo:timeline[i].tipo,
        material:timeline[i].material, buff:timeline[i].buff,
        horario:timeline[i].horarioStr, data:timeline[i].data,
        capaUrl:timeline[i].capaUrl,
        secondsToStart: timeline[i].inicio - now,
        topicoUrl:timeline[i].topicoUrl, isBackup:false
      };
    }
  }
  return { status:"off", programa:"Empire TV", message:"Programação encerrada. Até amanhã!", isBackup:true };
}

function buildPayload(item, seek) {
  return {
    status:"broadcasting", rowNum:item.rowNum, programa:item.programa,
    tipo:item.tipo, material:item.material, buff:item.buff,
    videoUrl:item.videoUrl, driveId:item.driveId,
    horario:item.horarioStr, data:item.data, duracao:item.duracao,
    inicio:item.inicio, seekOffset:Math.min(seek, item.duracao-1),
    topicoId:item.topicoId, topicoUrl:item.topicoUrl,
    capaUrl:item.capaUrl, isBackup:false
  };
}

function atualizarStatus(sheet, timeline, current) {
  try {
    if (!current || current.isBackup) return;
    var now = getSecondsToday();
    timeline.forEach(function(item) {
      var cellStatus  = sheet.getRange(item.rowNum, COL.STATUS+1);
      var statusAtual = String(cellStatus.getValue()).trim().toLowerCase();
      if (current.rowNum === item.rowNum) {
        if (statusAtual !== "transmitindo") cellStatus.setValue("Transmitindo");
      } else if (now >= item.fim) {
        if (statusAtual !== "finalizado") cellStatus.setValue("Finalizado");
      }
    });
  } catch(e) { Logger.log("Erro atualizarStatus: " + e); }
}

function getSecondsToday() {
  var now   = new Date();
  var local = new Date(now.toLocaleString("en-US", { timeZone: "America/Sao_Paulo" }));
  return (local.getHours()*3600) + (local.getMinutes()*60) + local.getSeconds();
}
